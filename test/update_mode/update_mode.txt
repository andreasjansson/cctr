====
cd $CCTR_FIXTURE_DIR/multi
cctr shifting.txt -u --no-color > /dev/null 2>&1 || true
cat shifting.txt

alpha
===
printf "a1\na2\na3\n"

====
cd $CCTR_FIXTURE_DIR/multi
cctr shifting.txt --no-color 2>&1 | tail -1
----
All 3 tests passed in {{ t }}s

====
cd $CCTR_FIXTURE_DIR/multi
